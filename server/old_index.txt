// server/index.js
import "dotenv/config";
import express from "express";
import multer from "multer";
import { conversational, buildSystem, getHistory, llm, pushManualTurn } from "./langchain.js";
import { v4 as uuid } from "uuid";
import {
  saveHistory,
  appendProjectMemory,
  readProjectMemory,
  ensureUploadDir,
  recordUploadedFile,
  listUploadedFiles
} from "./persist.js";
import path from "path";
import fs from "fs";
import { fileURLToPath } from "url";

const app = express();
app.use(express.json({ limit: "10mb" }));

// Multer for uploads (use diskStorage so big files don’t sit in RAM)
const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    const projectId = req.body.projectId || "sketchtiler";
    const dir = ensureUploadDir(projectId);
    cb(null, dir);
  },
  filename: (req, file, cb) => {
    const ts = Date.now();
    // sanitize basic name
    const safe = file.originalname.replace(/[^\w.\-]+/g, "_");
    cb(null, `${ts}-${safe}`);
  }
});
const upload = multer({
  storage,
  limits: { fileSize: 25 * 1024 * 1024 } // 25MB cap; adjust as you like
});

// Persist LangChain chat history helper
async function persistChat(projectId) {
  const h = getHistory(projectId);
  const msgs = await h.getMessages();
  const flat = msgs.map(m => ({
    role: m._getType() === "human" ? "user" : "assistant",
    content: m.content
  }));
  saveHistory(projectId, flat);
}

// -------- existing chat routes (unchanged) --------
app.post("/api/chat", async (req, res) => {
  try {
    const { projectId = "sketchtiler", userText = "" } = req.body;
    if (!userText.trim()) return res.status(400).json({ error: "userText is required" });

    const system = buildSystem(projectId);
    const responseText = await conversational.invoke(
      { input: userText, system },
      { configurable: { sessionId: projectId } }
    );

    await persistChat(projectId);
    res.json({ threadId: uuid(), text: responseText });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: String(e) });
  }
});

app.post("/api/chat-with-image", upload.single("image"), async (req, res) => {
  try {
    const { projectId = "sketchtiler", userText = "" } = req.body;
    if (!userText && !req.file) {
      return res.status(400).json({ error: "Provide userText and/or an image file." });
    }
    const mime = req.file?.mimetype || "image/png";
    const b64  = req.file ? fs.readFileSync(req.file.path).toString("base64") : null;

    const userContent = [];
    if (userText) userContent.push({ type: "text", text: userText });
    if (b64) {
      userContent.push({
        type: "image_url",
        image_url: { url: `data:${mime};base64,${b64}`, detail: "auto" } // detail optional
      });
    }

    await pushManualTurn(projectId, "user", userText || "[image]");
    const system = buildSystem(projectId);

    const result = await llm.invoke([
      { role: "system", content: system },
      { role: "user", content: userContent }
    ]);

    const assistantText =
      result?.content?.map?.(c => c?.text).join("") ||
      result?.content?.text ||
      String(result?.content || "");

    await pushManualTurn(projectId, "assistant", assistantText);
    await persistChat(projectId);

    res.json({ threadId: uuid(), text: assistantText });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: String(e) });
  }
});

// -------- NEW: upload files to store + summarize into memory --------
app.post("/api/memory/upload", upload.array("files", 8), async (req, res) => {
  try {
    const { projectId = "sketchtiler", note = "" } = req.body;
    const files = req.files || [];
    if (files.length === 0 && !note.trim()) {
      return res.status(400).json({ error: "Attach at least one file or provide a note." });
    }

    const saved = [];

    // Summarize each file (brief) and append to project memory
    for (const f of files) {
      const meta = {
        storedName: path.basename(f.path),
        originalName: f.originalname,
        size: f.size,
        mimetype: f.mimetype,
        uploadedAt: new Date().toISOString(),
      };
      recordUploadedFile(projectId, meta);
      saved.push(meta);

      // For small text-like files, try to read content directly (optional)
      let contentText = "";
      const isTextish = /^text\/|\/json$|\/xml$|\/csv$|\/yaml$|\/markdown$/.test(f.mimetype) ||
                        /\.(txt|md|json|csv|xml|yaml|yml)$/i.test(f.originalname);

      if (isTextish && f.size < 1024 * 1024 * 2) { // 2MB limit for direct read
        try { contentText = fs.readFileSync(f.path, "utf8").slice(0, 20000); } catch {}
      }

      // Ask the model for a compact summary/caption to store in memory
      const prompt = [
        { role: "system", content: "You are a concise archivist. Summarize files for future retrieval." },
        { role: "user", content: [
          { type: "text", text:
            `Summarize this file in 3-6 bullets for future use in SketchTiler. 
             Include: file purpose, key entities/rules, and how it could inform level design.
             Filenames: ${meta.originalName} (${meta.mimetype}, ${meta.size} bytes).` },
          ...(contentText
            ? [{ type: "text", text: `File preview:\n${contentText}` }]
            : [{ type: "text", text: "(Binary or large file; no inline preview.)" }])
        ]}
      ];
      const summary = await llm.invoke(prompt);
      const summaryText =
        summary?.content?.map?.(c => c?.text).join("") ||
        summary?.content?.text ||
        String(summary?.content || "");

      appendProjectMemory(
        projectId,
        `Uploaded file: ${meta.originalName}\nStored as: ${meta.storedName}\nType: ${meta.mimetype}, Size: ${meta.size} bytes\nSummary:\n${summaryText}`
      );
    }

    // Also append any freeform note the user included
    if (note && note.trim()) {
      appendProjectMemory(projectId, `Note with uploads:\n${note.trim()}`);
    }

    return res.json({ ok: true, saved });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: String(e) });
  }
});

// List stored files for a project
app.get("/api/memory/files", async (req, res) => {
  const projectId = req.query.projectId || "sketchtiler";
  const files = listUploadedFiles(projectId);
  res.json({ projectId, files });
});

// ---- Ingest a local tileset image + tilesheet spec into persistent memory ----
app.post("/api/memory/ingest-tileset", async (req, res) => {
  try {
    const {
      projectId = "sketchtiler",
      tilesetPath = "../tilemap.png",     // relative to /server/ folder
      specPath    = "../Tilesheet.txt"    // relative to /server/ folder
    } = req.body || {};

    const serverDir = path.dirname(fileURLToPath(import.meta.url));
    const tilesetAbs = path.resolve(serverDir, tilesetPath);
    const specAbs    = path.resolve(serverDir, specPath);

    if (!fs.existsSync(tilesetAbs)) return res.status(404).json({ error: `tileset not found: ${tilesetAbs}` });
    if (!fs.existsSync(specAbs))    return res.status(404).json({ error: `spec not found: ${specAbs}` });

    // 1) Copy files into the project upload vault
    const uploadDir = ensureUploadDir(projectId);
    const tsName = `${Date.now()}-${path.basename(tilesetAbs)}`;
    const spName = `${Date.now()}-${path.basename(specAbs)}`;

    fs.copyFileSync(tilesetAbs, path.join(uploadDir, tsName));
    fs.copyFileSync(specAbs,    path.join(uploadDir, spName));

    // 2) Record file metadata
    const tilesetStat = fs.statSync(path.join(uploadDir, tsName));
    const specStat    = fs.statSync(path.join(uploadDir, spName));

    recordUploadedFile(projectId, {
      storedName: tsName,
      originalName: path.basename(tilesetAbs),
      size: tilesetStat.size,
      mimetype: "image/png",
      uploadedAt: new Date().toISOString(),
    });
    recordUploadedFile(projectId, {
      storedName: spName,
      originalName: path.basename(specAbs),
      size: specStat.size,
      mimetype: "text/plain",
      uploadedAt: new Date().toISOString(),
    });

    // 3) Parse the spec text for canonical fields
    const rawSpec = fs.readFileSync(specAbs, "utf8");

    // crude parse that looks for numbers in lines
    const grab = (label) => {
      const re = new RegExp(label + "[^\\d]*(\\d+)[^\\d]+(\\d+)", "i");
      const m = rawSpec.match(re);
      return m ? [Number(m[1]), Number(m[2])] : null;
    };

    const sizePair   = grab("Tile size");              // [w, h]
    const spacePair  = grab("Space between tiles");    // [xspace, yspace]

    // totals may be on separate lines in your file; handle both 2-number or 1-number lines
    const totalH = (rawSpec.match(/Total tiles \(horizontal\)[^\d]*(\d+)/i)?.[1]) ?? null;
    const totalV = (rawSpec.match(/Total tiles \(vertical\)[^\d]*(\d+)/i)?.[1]) ?? null;
    const gridPair = totalH && totalV ? [Number(totalH), Number(totalV)] : null;

    const totalSheet = rawSpec.match(/Total tiles in sheet[^\d]*(\d+)/i)?.[1];

    // 4) Append a persistent, machine-friendly memory block
    const md = [
      "Tileset: Kenney Tiny Town (spaced atlas)",
      `Source: https://kenney.nl/assets/tiny-town`,
      `Image stored: ${tsName}`,
      `Spec stored: ${spName}`,
      "",
      "```tileset",
      `tile_size: ${sizePair ? `${sizePair[0]}x${sizePair[1]} px` : "unknown"}`,
      `spacing:   ${spacePair ? `${spacePair[0]}x${spacePair[1]} px` : "unknown"}`,
      `grid:      ${gridPair ? `${gridPair[0]}x${gridPair[1]} tiles` : "unknown"}`,
      `count:     ${totalSheet ? Number(totalSheet) : "unknown"}`,
      "```",
      "",
      "Usage notes:",
      "- Treat as a 16×16 grid with 1px gutters (if spec says so).",
      "- When slicing, include spacing so UVs don’t bleed.",
      "- Keep this tileset name stable: `kenney_tiny_town_spaced_16`."
    ].join("\n");

    appendProjectMemory(projectId, md);

    return res.json({
      ok: true,
      stored: { tileset: tsName, spec: spName },
      parsed: {
        tile_size: sizePair || null,
        spacing: spacePair || null,
        grid: gridPair || null,
        count: totalSheet ? Number(totalSheet) : null
      }
    });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: String(e) });
  }
});

// --- Generate a map from free-form text; returns strict JSON the UI can render ---
app.post("/api/map/generate", async (req, res) => {
  try {
    const { projectId = "sketchtiler", prompt = "" } = req.body || {};
    if (!prompt.trim()) return res.status(400).json({ error: "prompt is required" });

    // We’ll strongly steer the model to output JSON only
    const system = [
      "You convert natural language into a tilemap JSON for a 2D grid.",
      "Return ONLY valid minified JSON with keys: width, height, tiles[].",
      "Each tile: { x, y, col, row } where (col,row) index into the tileset atlas (0-based).",
      "Never include comments or extra text.",
      "If user asks for sizes, assume 16x16 tiles and 1px spacing unless stated otherwise."
    ].join(" ");

    const user = [
      "Create a tilemap from this request:",
      prompt,
      "",
      "Output JSON shape:",
      `{"width":W,"height":H,"tiles":[{"x":0,"y":0,"col":C,"row":R}, ...]}`
    ].join("\n");

    const result = await llm.invoke([
      { role: "system", content: system },
      { role: "user", content: user }
    ]);

    let text =
      result?.content?.map?.(c => c?.text).join("") ||
      result?.content?.text ||
      String(result?.content || "");

    // Try to recover the JSON if there’s accidental prose
    const match = text.match(/\{[\s\S]*\}$/);
    if (match) text = match[0];

    const data = JSON.parse(text); // throws if invalid → catch below

    // Basic shape checks
    if (
      typeof data.width !== "number" ||
      typeof data.height !== "number" ||
      !Array.isArray(data.tiles)
    ) {
      return res.status(400).json({ error: "Invalid JSON shape from model." });
    }

    return res.json({ ok: true, map: data });
  } catch (e) {
    console.error("map/generate error:", e);
    return res.status(500).json({ error: "Failed to generate map JSON." });
  }
});

// --- Return tileset URL + size/spacing inferred from persistent memory ---
app.get("/api/memory/tileset-meta", async (req, res) => {
  try {
    const projectId = req.query.projectId || "sketchtiler";
    const mem = readProjectMemory(projectId) || "";

    // pull `tile_size: 16x16 px` and `spacing: 1x1 px` from the memory block, if present
    const size = mem.match(/tile_size:\s*(\d+)\s*x\s*(\d+)/i);
    const spacing = mem.match(/spacing:\s*(\d+)\s*x\s*(\d+)/i);

    const tileWidth  = size ? Number(size[1]) : 16;
    const tileHeight = size ? Number(size[2]) : 16;
    const spaceX     = spacing ? Number(spacing[1]) : 1;
    const spaceY     = spacing ? Number(spacing[2]) : 1;

    // find the most recent PNG in uploads for this project
    const uploadDir = ensureUploadDir(projectId);
    const files = fs.readdirSync(uploadDir).filter(f => f.toLowerCase().endsWith(".png"));
    files.sort((a, b) => fs.statSync(path.join(uploadDir, b)).mtimeMs - fs.statSync(path.join(uploadDir, a)).mtimeMs);
    const latestPng = files[0] || null;

    // We expose uploads directory statically at /uploads
    const serverDir = path.dirname(fileURLToPath(import.meta.url));
    if (!app._uploadsMounted) {
      app.use("/uploads", express.static(path.join(serverDir, "data/uploads")));
      app._uploadsMounted = true;
    }

    const tilesetUrl = latestPng ? `/uploads/${encodeURIComponent(projectId)}/${encodeURIComponent(latestPng)}` : null;

    return res.json({
      ok: true,
      tilesetUrl,
      tileWidth,
      tileHeight,
      spaceX,
      spaceY
    });
  } catch (e) {
    console.error("tileset-meta error:", e);
    return res.status(500).json({ error: "Failed to read tileset meta." });
  }
});
